        async function handleImportExcel(e) {
            const file = e.target.files[0];
            if(!file) return;
            showLoading();
            const reader = new FileReader();
            reader.onload = async function(ev) {
                try {
                    const data = new Uint8Array(ev.target.result);
                    const workbook = XLSX.read(data, {type: 'array'});
                    const firstSheet = workbook.SheetNames[0];
                    const excelRows = XLSX.utils.sheet_to_json(workbook.Sheets[firstSheet]);
                    
                    if(excelRows.length === 0) throw new Error("Data kosong atau format salah.");
                    
                    let targetSheet = '';
                    if(activeMasterTab === 'part') targetSheet = 'Spareparts';
                    else if(activeMasterTab === 'service') targetSheet = 'Services';
                    else if(activeMasterTab === 'kategori') targetSheet = 'Kategori';
                    else if(activeMasterTab === 'pelanggan') targetSheet = 'Pelanggan';

                    const res = await Backend.run('importDataMassal', targetSheet, excelRows);
                    hideLoading(); showToast(res.message, res.success ? 'success' : 'error');
                    if(res.success) loadMasterData();
                } catch(err) { hideLoading(); showToast("Gagal memproses Excel: " + err.message, 'error'); }
                document.getElementById('file-import').value = ""; // reset
            };
            reader.readAsArrayBuffer(file);
        }

        function addTrxItemFromSelect() {
            const sel=document.getElementById('trx-item-select'); const qty=parseInt(document.getElementById('trx-item-qty').value)||1;
            if(!sel.value) return showToast('Pilih item dari dropdown!','error');
            const opt = sel.options[sel.selectedIndex]; const prc = parseInt(opt.dataset.price);
            
            // Check stok if it's a part
            if(sel.value.startsWith('prt_')) {
                const partData = rawParts.find(p => p['ID Part'] === sel.value.replace('prt_', ''));
                if(partData && partData.Stok < qty) return showToast(`Stok tidak mencukupi! Sisa stok: ${partData.Stok}`, 'error');
            }

            currentTrxItems.push({id:sel.value, name:opt.dataset.name, price:prc, qty:qty, subtotal:prc*qty});
            sel.value=""; document.getElementById('trx-item-qty').value="1"; renderTrxItemsUI();
        }

        async function previewLaporanRentang() {

        async function previewMasterLaporan(sheetName) {
            showLoading();
            try {
                const data = await Backend.run('getTableData', sheetName);
                hideLoading();
                
                if(data.length === 0) return showToast("Tidak ada data untuk dicetak", "error");
                
                let thHtml = '';
                let tbHtml = '';
                const headers = Object.keys(data[0]);
                
                headers.forEach(h => { thHtml += `<th class="border border-slate-400 p-2 text-left">${h}</th>`; });
                
                data.forEach(row => {
                    tbHtml += '<tr>';
                    headers.forEach(h => {
                        let val = row[h];
                        if(h.toLowerCase().includes('harga') || h.toLowerCase().includes('biaya')) val = formatRp(val);
                        if(h.toLowerCase().includes('tanggal') && val) {
                            try { val = formatTgl(val); } catch(e){}
                        }
                        tbHtml += `<td class="border border-slate-400 p-2">${val !== undefined ? val : ''}</td>`;
                    });
                    tbHtml += '</tr>';
                });

                document.getElementById('print-area').innerHTML = `
                    <div class="text-center mb-6">
                        <h2 class="text-2xl font-bold uppercase underline">Laporan Data ${sheetName}</h2>
                        <p class="text-sm text-slate-600">Dicetak pada: ${formatTgl(new Date().toISOString())}</p>
                    </div>
                    <table class="w-full text-sm border-collapse border border-slate-400 mb-8">
                        <thead class="bg-slate-100"><tr>${thHtml}</tr></thead>
                        <tbody>${tbHtml}</tbody>
                    </table>`;
                
                document.getElementById('print-actions').innerHTML = `
                    <button onclick="window.print()" class="flex-1 bg-blue-600 text-white py-3 rounded-lg text-sm font-bold hover:bg-blue-700 flex items-center justify-center gap-2 shadow-md"><i class="ph-bold ph-printer"></i> Cetak / Simpan PDF</button>
                    <button onclick="closeModal('modal-receipt')" class="flex-1 bg-slate-100 border border-slate-300 py-3 rounded-lg text-sm font-bold hover:bg-slate-200 text-slate-800">Tutup Preview</button>
                `;
                
                document.getElementById('receipt-container').className = "bg-white text-black w-full max-w-4xl rounded-xl p-5 md:p-8 relative z-10 shadow-2xl my-8";
                openModal('modal-receipt');
            } catch(e) {
                hideLoading();
                showToast("Gagal memuat laporan: " + e.message, "error");
            }
        }

        function exportLaporanExcel() {
            if(currentFilteredTrx.length === 0) return showToast("Tidak ada data untuk diexport", "error");
            // Buat copy data yang rapi untuk Excel
            const cleanData = currentFilteredTrx.map(t => {
                return {
                    'ID Transaksi': t['ID Transaksi'], 'Tanggal': formatTgl(t.Tanggal), 'No Polisi': t['No Polisi'], 'Pelanggan': t.Pelanggan,
                    'Rincian Transaksi': t['Layanan & Sparepart'], 'Total Biaya (Rp)': t['Total Biaya'], 'Status Pembayaran': t.Status
                };
            });
            const ws = XLSX.utils.json_to_sheet(cleanData); const wb = XLSX.utils.book_new();
            XLSX.utils.book_append_sheet(wb, ws, "Laporan_Trx"); XLSX.writeFile(wb, `Laporan_Transaksi_${new Date().getTime()}.xlsx`);
            showToast("Laporan Excel berhasil diunduh");
        }

        async function exportExcelData(sheetName) {
            showLoading();
            try {
                const data = await Backend.run('getTableData', sheetName);
                if(data.length===0) return showToast('Tidak ada data','error');
                const ws = XLSX.utils.json_to_sheet(data); const wb = XLSX.utils.book_new();
                XLSX.utils.book_append_sheet(wb, ws, sheetName); XLSX.writeFile(wb, `Master_${sheetName}_${new Date().getTime()}.xlsx`);
                showToast('Berhasil download Excel (.xlsx)');
            } catch(e) { showToast('Gagal Export: '+e.message, 'error'); } finally { hideLoading(); }
        }
